# 1. Khởi tạo Git ở thư mục máy tính
git init

# 2. Thêm file README.md
git add README.md

# 3. Lưu bản commit đầu tiên
git commit -m "docs: khoi tao README mo ta du an"

# 4. Chuyển nhánh chính thành main
git branch -M main

# 5. Liên kết với kho lưu trữ GitHub của bạn
git remote add origin https://github.com/nch-23013124/Sports-ecommerce.git

# 6. Đẩy dữ liệu lên GitHub
git push -u origin main
